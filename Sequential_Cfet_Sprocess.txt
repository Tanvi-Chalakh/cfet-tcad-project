proc WriteBND {} {
    global count

    if { $count < 10 } {
        struct tdr.bnd=n@node@_0${count}
    } else {
        struct tdr.bnd=n@node@_${count}
    }

    puts "STAGE: WriteBND #$count done"
    set count [expr $count+1]
}

# Thermal step: always logged; runs diffuse only if run=1.
# eq1000C = equivalent time at 1000 C with a generic Ea=3.5 eV
# (a bookkeeping number, not a physical model).
set TB_LOG [open "n@node@_thermal_budget.log" w]
set TB_EQ  0.0
proc THERMAL {label T t run} {
    global TB_LOG TB_EQ
    set k 8.617e-5
    set eq [expr {$t*exp(-3.5/$k*(1.0/($T+273.15)-1.0/1273.15))}]
    set TB_EQ [expr {$TB_EQ+$eq}]
    puts $TB_LOG [format "%-30s T=%6.1f C  t=%g s  eq1000C=%.3e s  cum=%.3e s  run=%d" $label $T $t $eq $TB_EQ $run]
    if { $run } {
        diffuse time=${t}<s> temp=$T !stress.relax
    }
}

puts $TB_LOG "Top channel orientation (record only): Si($topOrient)"

# Mesh-line helper for x: skips a line if one already exists within ltol
set ltol 0.1e-3
set lnlist(x) {}
proc LN {dir loc sp {tag ""}} {
    global lnlist ltol
    foreach xx $lnlist($dir) {
        if { abs($loc - $xx) < $ltol } {
            puts "LN: skipping $dir line at $loc (already have one at $xx)"
            return
        }
    }
    lappend lnlist($dir) $loc
    if { $tag eq "" } {
        line $dir loc=$loc spacing=$sp
    } else {
        line $dir loc=$loc spacing=$sp tag=$tag
    }
}

# If a geometric line coincides with a doping-transition line, give the

    global P0dop P1dop ztol dz dzf
    if { abs($loc - $P0dop) < $ztol || abs($loc - $P1dop) < $ztol } {
        return $dzf

    return $dz
}

# Fill: replace materials in a brick with a new material/region.
proc Fill {x0 y0 z0 x1 y1 z1 repl mat reg} {
    polyhedron name=blk brick= [list $x0 $y0 $z0 $x1 $y1 $z1]
    insert polyhedron=blk replace.materials= $repl \
           new.material=$mat new.region=$reg
    PolyHedronClear
}

# RoundSlab: a nanosheet-like slab with fully rounded ends (box + 2 cylinders
# along z). x0..x1 is the thickness, y0..y1 the outer width (incl. round ends).
proc RoundSlab {x0 x1 y0 y1 repl mat reg} {
    global Zlo Zhi Lext

    set R   [expr 0.5*($x1-$x0)]
    set yA  [expr $y0+$R]
    set yB  [expr $y1-$R]
    set xm  [expr 0.5*($x0+$x1)]

    if { $yB > $yA } {
        polyhedron name=slab brick= [list $x0 $yA $Zlo $x1 $yB $Zhi]
        insert polyhedron=slab replace.materials= $repl \
               new.material=$mat new.region=$reg
        PolyHedronClear
    }

    foreach yc [list $yA $yB] {
        PolyHedronCylinder slab $Zlo $yc $xm $R $Lext Z
        insert polyhedron=slab replace.materials= $repl \
               new.material=$mat new.region=$reg
        PolyHedronClear
    }
}

# Helper: put region r2 into the same contact as r1 (common contact).
# If "add" is not accepted by your version, r2 gets its own contact (alt2);
# then tie the two in SDevice.
proc MakeCommon {cname r1 r2 alt2} {
    contact name=$cname region=$r1 replace
    if { [catch {contact name=$cname region=$r2 add} errAdd] } {
        puts "WARNING: could not add $r2 to contact $cname ($errAdd); using separate contact $alt2"
        contact name=$alt2 region=$r2 replace
    }
}

}

    contact name=drain_n  region=dMetal_n replace

} elseif { $contactMode == 1 } {
    contact name=source_n region=sMetal_n replace
    contact name=source_p region=sMetal_p replace
    MakeCommon drain dMetal_n dMetal_p drain_p
} else {
    MakeCommon source sMetal_n sMetal_p source_p
    MakeCommon drain  dMetal_n dMetal_p drain_p
}
