#setdep @node|genStackedNanoslab@

*=====================================================================*
* CFET ELECTRICAL CHARACTERISATION - SENTaurus DEVICE
*=====================================================================*
* Simulates the electrical behaviour of the vertically stacked CFET:
* bottom nFET and top pFET.
*
* Workbench parameters:
*   contactMode : 0 = six separate terminals; 1 = common source/drain
*   VDRAIN     : drain bias magnitude
*   VGATE      : positive gate-sweep limit
*   WF         : gate-metal workfunction
*   Threads    : number of solver threads
*
* The input structure is the final doped TDR generated by SProcess.
* The top pFET channel is assigned Si0.8Ge0.2 through MoleFraction.
*=====================================================================*

#define _WF_ @WF@

*=====================================================================*
* CHANNEL-CARRIER INTEGRATION WINDOWS
* Coordinates are derived from the SProcess geometry.
* Units: um
*=====================================================================*

*---------------------------------------------------------------
* CHANNEL-CARRIER INTEGRATION WINDOWS
* Stack-dependent coordinates derived from SProcess
* Units: um
*---------------------------------------------------------------

#define _NSTACKB_ 1
#define _NSTACKT_ 1

#define _TSIGE_ 10.0e-3
#define _TMID_  20.0e-3

#define _XN0_ @<-0.5*H>@
#define _XN1_ @<0.5*H+(_NSTACKB_-1)*(H+_TSIGE_)>@

#define _XP0_ @<0.5*H+(_NSTACKB_-1)*(H+_TSIGE_)+_TMID_>@
#define _XP1_ @<0.5*H+(_NSTACKB_-1)*(H+_TSIGE_)+_TMID_+H+(_NSTACKT_-1)*(H+_TSIGE_)>@

#define _YW_  @<0.5*W>@
#define _ZW_  @<0.5*L>@

*=====================================================================*
* ELECTRICAL CONTACT DEFINITIONS
*=====================================================================*
* The gate uses the specified metal workfunction.
* contactMode=1 uses common source/drain terminals.
* contactMode=0 keeps nFET and pFET source/drain terminals separate.
* No substrate terminal is defined in this SProcess structure.
Electrode {
    { Name="gate" Voltage=0.0 Workfunction=_WF_ }

#if "@contactMode@" == "1"
    { Name="source" Voltage=0.0 }
    { Name="drain"  Voltage=0.0 }
#else
    { Name="source_n" Voltage=0.0 }
    { Name="drain_n"  Voltage=0.0 }
    { Name="source_p" Voltage=0.0 }
    { Name="drain_p"  Voltage=0.0 }
#endif
}

*=====================================================================*
* INPUT / OUTPUT FILES
*=====================================================================*
* Grid points to the final doped SProcess structure, not the intermediate
* mesh TDR. The latter may not contain the final doping/material mapping.
File {
    Grid    = "n@node|genStackedNanoslab@_CFET_fps.tdr"
    Plot    = "n@node@_des.tdr"
    Output  = "n@node@_des.log"
    Current = "n@node@_des.plt"
    Parameter = "@parameter@"
}

*=====================================================================*
* GLOBAL SEMICONDUCTOR PHYSICS
*=====================================================================*
* OldSlotboom models band-gap narrowing in heavily doped regions.
* PhuMob provides doping-dependent carrier mobility.
* Enormal accounts for mobility degradation due to the normal electric
* field at semiconductor interfaces.
* HighFieldSaturation accounts for mobility/velocity saturation at
* high lateral electric fields.
* SRH and Auger describe carrier-generation/recombination mechanisms.
Physics {
    EffectiveIntrinsicDensity(
        BandGapNarrowing(OldSlotboom)
    )

    Mobility(
        PhuMob
        Enormal
        HighFieldSaturation
    )

    Recombination(
        SRH(DopingDependence)
        Auger
    )
}

*=====================================================================*
* TOP pFET CHANNEL COMPOSITION
*=====================================================================*
* Defines Si(1-x)Ge(x) with x=0.2 for the top pFET channel.
* The corresponding SProcess region must be SiliconGermanium.
* If the TDR already supplies a Ge mole-fraction field, check that this
* material-level definition does not conflict with that field.
Physics(Material="SiliconGermanium") {
    MoleFraction(
        xFraction = 0.2
    )
}

*=====================================================================*
* OUTPUT QUANTITIES FOR POST-PROCESSING
*=====================================================================*
* Requests carrier densities, current densities, electrostatic
* quantities, mobility, band edges and quasi-Fermi levels for inspection
* in SVisual. Plot requests do not, by themselves, activate a model.
Plot {
    eDensity
    hDensity
    eCurrent/Vector
    hCurrent/Vector
    TotalCurrent/Vector

    Potential
    ElectricField/Vector
    Doping
    SpaceCharge

    eMobility
    hMobility

    BandGap
    Affinity
    BandGapNarrowing
    EffectiveBandGap
    EffectiveIntrinsicDensity

    ConductionBand
    ValenceBand

    eQuasiFermi
    hQuasiFermi
}

*=====================================================================*
* CHANNEL CARRIER COUNT
*=====================================================================*
* Integrates electron density over the bottom nFET channel window and
* hole density over the top pFET channel window.
* The window coordinates are specified above in the SProcess frame.
CurrentPlot {
    eDensity (
        Integrate(
            Window[(_XN0_ -_YW_ -_ZW_)
                   (_XN1_  _YW_  _ZW_)]
        )
    )

    hDensity (
        Integrate(
            Window[(_XP0_ -_YW_ -_ZW_)
                   (_XP1_  _YW_  _ZW_)]
        )
    )
}

*=====================================================================*
* NUMERICAL METHODS AND CONVERGENCE CONTROLS
*=====================================================================*
* UCS sets the coordinate system used by the solver.
* ILS selects the iterative linear solver; ILSrc configures GMRES and
* ILUT preconditioning.
* RelErrControl, Digits and Notdamped affect nonlinear convergence.
* Number_of_Threads is supplied through the Workbench parameter.
Math {
    CoordinateSystem { UCS }

    -CheckUndefinedModels
    AutoOrientationSmoothingDistance = -1
    GeometricDistances

    Extrapolate
    Derivative

    Method = ILS
    RhsMin = 1e-12
    Iterations = 20

    RelErrControl
    Digits = 5
    Notdamped = 100

    ILSrc = "
    set(1){
        iterative(gmres(1000),
            tolrel=1e-10,
            tolunprec=1e-8,
            tolabs=0,
            maxit=200);
        preconditioning(ilut(0.001,-1),left);
        ordering(symmetric=nd,nonsymmetric=mpsilst);
        options(compact=yes,linscale=0,
            refineresidual=8,verbose=0);
    };
    "

    wallclock
    ExitOnFailure
    Number_of_Threads = @Threads@

    CurrentPlot(IntegrationUnit=cm)
    CurrentWeighting
}

*=====================================================================*
* SOLUTION SEQUENCE
*=====================================================================*
Solve {

    *-----------------------------------------------------------------*
    * 1. EQUILIBRIUM INITIALISATION
    *-----------------------------------------------------------------*
    * First solve Poisson, then solve the coupled electron-hole system
    * to establish the equilibrium state before applying terminal bias.
    Coupled(
        Iterations=100
        LineSearchDamping=1e-3
    ) {
        Poisson
    }

    Coupled(
        Iterations=100
        LineSearchDamping=1e-3
    ) {
        Poisson Electron Hole
    }

    Save(FilePrefix="n@node@_eq")

    *-----------------------------------------------------------------*
    * 2. APPLY DRAIN BIAS
    *-----------------------------------------------------------------*
    * In common-terminal mode, bias the shared drain.
    * In separate-terminal mode, apply +VDRAIN to nFET and -VDRAIN
    * to pFET so that both devices are biased with the intended polarity.
    Quasistationary(
        InitialStep=0.05
        Increment=1.5
        MinStep=1e-5
        MaxStep=0.25

#if "@contactMode@" == "1"
        Goal {
            Name="drain"
            Voltage=@VDRAIN@
        }
#else
        Goal {
            Name="drain_n"
            Voltage=@VDRAIN@
        }

        Goal {
            Name="drain_p"
            Voltage=-@VDRAIN@
        }
#endif

    ) {
        Coupled(
            Iterations=50
            LineSearchDamping=1
        ) {
            Poisson Electron Hole
        }
    }

    Save(FilePrefix="n@node@_Vd")

    *-----------------------------------------------------------------*
    * 3. SET GATE TO THE NEGATIVE SWEEP STARTING POINT
    *-----------------------------------------------------------------*
    * Moves the gate from 0 V to -VGATE while holding the drain bias.
    * This establishes the starting point for the negative-to-positive
    * transfer-characteristic sweep.
    Quasistationary(
        InitialStep=0.05
        Increment=1.5
        MinStep=1e-7
        MaxStep=0.1

        Goal {
            Name="gate"
            Voltage=-@VGATE@
        }

    ) {
        Coupled(
            Iterations=50
            LineSearchDamping=1
        ) {
            Poisson Electron Hole
        }
    }

    Save(FilePrefix="n@node@_VgNeg")

    *-----------------------------------------------------------------*
    * 4. GATE SWEEP: -VGATE TO 0 V
    *-----------------------------------------------------------------*
    * Quasistationary stepping ramps the gate voltage to zero.
    * A smaller maximum step and increment are used for the sweep.
    Quasistationary(
        InitialStep=1e-2
        Increment=1.3
        MinStep=1e-7
        MaxStep=0.02

        Goal {
            Name="gate"
            Voltage=0.0
        }

    ) {
        Coupled(Iterations=50) {
            Poisson Electron Hole
        }
    }

    Save(FilePrefix="n@node@_VgMid")

    *-----------------------------------------------------------------*
    * 5. GATE SWEEP: 0 V TO +VGATE
    *-----------------------------------------------------------------*
    * Continues the transfer sweep from zero to the positive gate limit.
    Quasistationary(
        InitialStep=1e-2
        Increment=1.3
        MinStep=1e-7
        MaxStep=0.02

        Goal {
            Name="gate"
            Voltage=@VGATE@
        }

    ) {
        Coupled(Iterations=50) {
            Poisson Electron Hole
        }
    }

    Save(FilePrefix="n@node@_VgPos")
}
